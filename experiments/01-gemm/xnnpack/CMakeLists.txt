cmake_minimum_required(VERSION 3.24)

project(
    xnnpack_gemm_demo
    LANGUAGES CXX
)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

#
# Repository root:
#
# experiments/01-gemm/xnnpack
#   -> ../../..
#
get_filename_component(
    AI_COMPILER_LAB_ROOT
    "${CMAKE_CURRENT_SOURCE_DIR}/../../.."
    ABSOLUTE
)

set(
    XNNPACK_SOURCE_DIR
    "${AI_COMPILER_LAB_ROOT}/third_party/XNNPACK"
)

set(
    XNNPACK_BUILD_DIR
    "${XNNPACK_SOURCE_DIR}/build/local"
)

#
# Fail early if the expected XNNPACK build is missing.
#
foreach(required_file
    "${XNNPACK_SOURCE_DIR}/include/xnnpack.h"
    "${XNNPACK_BUILD_DIR}/libXNNPACK.a"
    "${XNNPACK_BUILD_DIR}/libxnnpack-microkernels-prod.a"
)
    if(NOT EXISTS "${required_file}")
        message(
            FATAL_ERROR
            "Required XNNPACK artifact not found:\n"
            "  ${required_file}\n"
            "Run scripts/build-xnnpack.sh first."
        )
    endif()
endforeach()

#
# XNNPACK public header includes pthreadpool.h.
#
set(
    PTHREADPOOL_INCLUDE_DIR
    "${XNNPACK_BUILD_DIR}/pthreadpool-source/include"
)

#
# Libraries generated by the XNNPACK local build.
#
find_library(
    PTHREADPOOL_LIBRARY
    NAMES pthreadpool
    PATHS
        "${XNNPACK_BUILD_DIR}/pthreadpool"
    NO_DEFAULT_PATH
)

find_library(
    CPUINFO_LIBRARY
    NAMES cpuinfo
    PATHS
        "${XNNPACK_BUILD_DIR}/cpuinfo"
    NO_DEFAULT_PATH
)

if(NOT PTHREADPOOL_LIBRARY)
    message(
        FATAL_ERROR
        "libpthreadpool not found under:"
        " ${XNNPACK_BUILD_DIR}/pthreadpool"
    )
endif()

if(NOT CPUINFO_LIBRARY)
    message(
        FATAL_ERROR
        "libcpuinfo not found under:"
        " ${XNNPACK_BUILD_DIR}/cpuinfo"
    )
endif()

add_executable(
    xnnpack-gemm
    app/main.cpp
)

target_include_directories(
    xnnpack-gemm
    PRIVATE
        "${XNNPACK_SOURCE_DIR}/include"
        "${PTHREADPOOL_INCLUDE_DIR}"
)

#
# libXNNPACK.a itself references:
#
#   production microkernels
#   pthreadpool
#   cpuinfo
#   libm
#
# Because these are static libraries, link order matters.
#
target_link_libraries(
    xnnpack-gemm
    PRIVATE
        "${XNNPACK_BUILD_DIR}/libXNNPACK.a"
        "${XNNPACK_BUILD_DIR}/libxnnpack-microkernels-prod.a"
        "${PTHREADPOOL_LIBRARY}"
        "${CPUINFO_LIBRARY}"
        m
        pthread
)

target_compile_options(
    xnnpack-gemm
    PRIVATE
        -Wall
        -Wextra
        -Wpedantic
)

message(STATUS "AI compiler lab : ${AI_COMPILER_LAB_ROOT}")
message(STATUS "XNNPACK source  : ${XNNPACK_SOURCE_DIR}")
message(STATUS "XNNPACK build   : ${XNNPACK_BUILD_DIR}")
message(STATUS "pthreadpool     : ${PTHREADPOOL_LIBRARY}")
message(STATUS "cpuinfo         : ${CPUINFO_LIBRARY}")
